function renderForecast(forecast){ const strip = $('forecastStrip'); strip.innerHTML = ''; forecast.properties.periods.slice(0, 14).forEach((p, index) => { const card = document.createElement('div'); card.className = 'fc-card'; const dayLabel = index === 0 ? 'Now' : p.name; card.innerHTML = `
${dayLabel}
${p.isDaytime ? 'Day' : 'Night'}
${p.temperature}°${p.temperatureUnit}
${p.shortForecast}
Wind${p.windSpeed}
Precip${p.probabilityOfPrecipitation && p.probabilityOfPrecipitation.value != null ? p.probabilityOfPrecipitation.value + '%' : '—'}
`; strip.appendChild(card); }); }
